fun isPrime n =
  if n < 2 then false
  else
    let
      fun checkDivisor d =
        if d * d > n then true
        else if n mod d = 0 then false
        else checkDivisor (d + 1)
    in
      checkDivisor 2
    end;

fun findPair n p =
  if p >= n then ()
  else if isPrime p andalso isPrime (n - p) then
    print (Int.toString p ^ " + " ^ Int.toString (n - p) ^ " = " ^ Int.toString n ^ "\n")
  else
    findPair n (p + 1);

(* Main goldbach function with error checking *)
fun goldbach n =
  if n <= 2 then
    print "Error: integer must be greater than 2\n"
  else if n mod 2 <> 0 then
    print "Error: integer must be even\n"
  else
    findPair n 2;

goldbach 28;
goldbach 6;
goldbach 100;
goldbach 3;
goldbach 2;
goldbach ~4;
